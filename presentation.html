A multi-layer perceptron is the simplest neural network architecture: input → one hidden layer → output. Every neuron is connected to every neuron in the next layer; training adjusts the strength of these connections so the network maps pixels to the correct digit.

What is the structure of the model?

784 pixel intensities flow into 256 hidden neurons (ReLU), then into 10 output neurons whose scores are turned into probabilities by softmax. The diagram is schematic - each dot stands for many neurons.

LayerNeuronsWeight matrixActivation
Input layer784none— raw pixels
Hidden layer256784 × 256ReLU
Output layer10256 × 10softmax → argmax
Total trainable parameters
203,530
weights + biases
Mathematical form
y = softmax( ReLU( xW₁ + b₁ ) W₂ + b₂ )
How one neuron works: it computes a weighted sum of its inputs (the weights were learned from data), adds a bias, then applies an activation function (ReLU). Without ReLU, stacking layers would collapse into a single linear model - the activation is what lets the network learn curved decision boundaries.

What to say: the training loss keeps falling toward 0, but the validation loss bottoms out at epoch 11 and rises again afterwards. The widening gap is overfitting - the model memorises the training data and generalises worse. This is why the deployed model uses early_stopping, which stops training near the best validation point instead of running all epochs.

Code that creates and trains the model

Complete, runnable code. The first block trains the model and saves it; the second shows how the trained network runs predictions in the browser on the demo page (weights exported once, forward pass in JavaScript).

backend/train_sklearn.py - load MNIST, create and fit the model

python train_sklearn.py --hidden 256 -> trains, evaluates (~98% test accuracy), saves models/mnist_mlp.joblib

import argparse
import gzip
import os
import struct
import time

import joblib
import numpy as np
from sklearn.metrics import accuracy_score
from sklearn.neural_network import MLPClassifier

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
RAW_DIR = os.path.join(BASE_DIR, "data", "MNIST", "raw")
MODEL_PATH = os.path.join(BASE_DIR, "models", "mnist_mlp.joblib")


def read_idx_images(path):
    with gzip.open(path, "rb") if path.endswith(".gz") else open(path, "rb") as f:
        magic, num, rows, cols = struct.unpack(">IIII", f.read(16))
        assert magic == 2051, f"bad image magic {magic}"
        buf = f.read()
    arr = np.frombuffer(buf, dtype=np.uint8).reshape(num, rows * cols)
    return arr.astype(np.float32) / 255.0


def read_idx_labels(path):
    with gzip.open(path, "rb") if path.endswith(".gz") else open(path, "rb") as f:
        magic, num = struct.unpack(">II", f.read(8))
        assert magic == 2049, f"bad label magic {magic}"
        buf = f.read()
    return np.frombuffer(buf, dtype=np.uint8).copy()


def main(hidden, max_iter, alpha):
    print("Loading MNIST from", RAW_DIR)
    X_train = read_idx_images(os.path.join(RAW_DIR, "train-images-idx3-ubyte"))
    y_train = read_idx_labels(os.path.join(RAW_DIR, "train-labels-idx1-ubyte"))
    X_test = read_idx_images(os.path.join(RAW_DIR, "t10k-images-idx3-ubyte"))
    y_test = read_idx_labels(os.path.join(RAW_DIR, "t10k-labels-idx1-ubyte"))
    print(
        f"train: {X_train.shape}  test: {X_test.shape}  "
        f"class distribution: {np.bincount(y_train)}"
    )

    model = MLPClassifier(
        hidden_layer_sizes=hidden,
        activation="relu",
        solver="adam",
        alpha=alpha,
        batch_size=256,
        learning_rate_init=1e-3,
        max_iter=max_iter,
        early_stopping=True,
        n_iter_no_change=5,
        validation_fraction=0.1,
        random_state=42,
        verbose=True,
    )

    print(f"Training MLPClassifier hidden={hidden} max_iter={max_iter} alpha={alpha} ...")
    start = time.time()
    model.fit(X_train, y_train)
    print(f"Training finished in {time.time() - start:.1f}s "
          f"after {model.n_iter_} iterations")

    y_pred = model.predict(X_test)
    acc = accuracy_score(y_test, y_pred)
    print(f"Test accuracy: {acc:.4f}")

    os.makedirs(os.path.dirname(MODEL_PATH), exist_ok=True)
    joblib.dump(model, MODEL_PATH)
    print(f"Model saved to {MODEL_PATH}")


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Train an MLPClassifier on MNIST")
    parser.add_argument("--hidden", type=str, default="256")
    parser.add_argument("--max-iter", type=int, default=60)
    parser.add_argument("--alpha", type=float, default=1e-4)
    args = parser.parse_args()
    layers = tuple(int(n) for n in args.hidden.split(",") if n.strip())
    main(layers or (256,), args.max_iter, args.alpha)

index.html (demo page) - forward pass + preprocessing in JavaScript

runs 100% in the browser, no server; preprocessing mirrors the Python pipeline

    function forward(x) {
      const { W1, b1, W2, b2 } = model;
      const h = new Float32Array(256);
      for (let k = 0; k < 256; k++) {
        let s = b1[k];
        const row = k * 784;
        for (let i = 0; i < 784; i++) s += x[i] * W1[row + i];
        h[k] = s > 0 ? s : 0;
      }
      const logits = new Float32Array(10);
      for (let j = 0; j < 10; j++) {
        let s = b2[j];
        const row = j * 256;
        for (let k = 0; k < 256; k++) s += h[k] * W2[row + k];
        logits[j] = s;
      }
      let m = logits[0];
      for (let j = 1; j < 10; j++) if (logits[j] > m) m = logits[j];
      let sum = 0;
      for (let j = 0; j < 10; j++) sum += (logits[j] = Math.exp(logits[j] - m));
      for (let j = 0; j < 10; j++) logits[j] /= sum;
      return logits;
    }

    /* ---------- preprocessing (mirrors the original backend) ---------- */
    function cornerMedianLight() {
      const patch = 5, vals = [];
      const spots = [[0, 0], [DISPLAY - patch, 0], [0, DISPLAY - patch], [DISPLAY - patch, DISPLAY - patch]];
      const d = ctx.getImageData(0, 0, DISPLAY, DISPLAY).data;
      for (const [x0, y0] of spots)
        for (let y = y0; y < y0 + patch; y++)
          for (let x = x0; x < x0 + patch; x++) vals.push(d[(y * DISPLAY + x) * 4]);
      vals.sort((a, b) => a - b);
      return vals[vals.length >> 1] > 128;
    }

    function toModelInput() {
      const bgLight = cornerMedianLight();
      const tmp = document.createElement("canvas");
      tmp.width = 28; tmp.height = 28;
      const c = tmp.getContext("2d", { willReadFrequently: true });
      c.imageSmoothingEnabled = true;
      c.imageSmoothingQuality = "high";
      c.drawImage(canvas, 0, 0, 28, 28);
      const px = c.getImageData(0, 0, 28, 28).data;
      const x = new Float32Array(784);
      for (let i = 0; i < 784; i++) {
        const v = px[i * 4] / 255;
        x[i] = bgLight ? 1 - v : v;
      }
      return x;
    }
